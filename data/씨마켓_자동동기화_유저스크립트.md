# 씨마켓 → 포털 자동 동기화 (유저스크립트) · v1.1

씨마켓은 로그인 뒤에 있는 데이터라 **서버가 직접 못 읽습니다.**
이미 로그인해 있는 **내 브라우저**가 대신 읽어 포털로 보내는 구조입니다.
아이디·비밀번호는 어디에도 저장되지 않습니다.

## v1.1 에서 고친 것

씨마켓은 SPA 라 화면을 옮겨도 **페이지가 새로 로드되지 않습니다.**
v1.0 은 로드 직후 20초만 보고 포기해서, 그 사이 「나의 거래 요약」이 아니면 그냥 끝나버렸습니다.

- **계속 지켜봅니다** — 15초마다 확인, 요약 화면이 보이면 그때 보냄
- **화면 이동도 감지** — SPA 내부 이동(주소만 바뀌는 경우)도 잡습니다
- **콘솔에 기록** — `[씨마켓동기화]` 로 무슨 일이 있었는지 남깁니다. 안 될 때 원인을 바로 볼 수 있습니다

## 설치 / 교체

이미 v1.0 을 넣으셨다면 **그 스크립트를 열어 내용을 전부 지우고** 아래로 바꾸신 뒤 `Ctrl + S`.

1. Tampermonkey 아이콘 → **대시보드**
2. `씨마켓 → FITI 업무포털 동기화` 클릭
3. 편집창 전체 선택(`Ctrl+A`) → 삭제 → 아래 코드 붙여넣기 → `Ctrl+S`

## 코드

```javascript
// ==UserScript==
// @name         씨마켓 → FITI 업무포털 동기화
// @namespace    fiti.procurement
// @version      1.1
// @description  씨마켓 나의 거래 요약 숫자를 업무포털로 자동 전송
// @match        https://www.c-market.net/*
// @run-at       document-idle
// @grant        none
// ==/UserScript==
(function () {
  'use strict';

  var PORTAL = 'https://fiti-procurement-portal.vercel.app';
  var KEY_LS = 'fiti.bidKey';
  var SIG_LS = 'fiti.bidSig';
  var busy = false;

  function log() {
    var a = ['[씨마켓동기화]'].concat([].slice.call(arguments));
    console.log.apply(console, a);
  }

  function num(s) {
    var m = String(s).replace(/,/g, '').match(/-?\d+/);
    return m ? parseInt(m[0], 10) : 0;
  }

  function read() {
    var T = document.body.innerText;
    if (T.indexOf('나의 거래 요약') < 0) return null;

    function after(label) {
      var i = T.indexOf(label);
      if (i < 0) return 0;
      var m = T.slice(i + label.length, i + label.length + 40).match(/-?[\d,]+/);
      return m ? num(m[0]) : 0;
    }

    var kpi = {
      inProgress: after('진행중'),
      closed: after('전체 마감'),
      contracts: after('총 계약'),
      cardPending: after('카드 결제 대기'),
      invoices: after('계산서 및 영수증')
    };
    if (!kpi.closed && !kpi.contracts) return null;   // 아직 안 그려짐

    var reasons = [], total = 0;
    var fi = T.indexOf('유찰 사유');
    if (fi >= 0) {
      var blk = T.slice(fi, fi + 1200).split('\n').map(function (x) { return x.trim(); }).filter(Boolean);
      var tm = blk.join(' ').match(/총\s*유찰\s*([\d,]+)\s*건/);
      if (tm) total = num(tm[1]);
      for (var i = 0; i < blk.length; i++) {
        var m = blk[i].match(/^(.+?)\s+([\d,]+)건\s*[·.]\s*[\d.]+%$/);
        if (m) { reasons.push({ label: m[1].trim(), count: num(m[2]) }); continue; }
        if (/건\s*[·.]\s*[\d.]+%/.test(blk[i]) && blk[i - 1] && !/유찰|총/.test(blk[i - 1])) {
          var c = blk[i].match(/([\d,]+)건/);
          if (c) reasons.push({ label: blk[i - 1], count: num(c[1]) });
        }
      }
    }
    if (!total) total = reasons.reduce(function (s, r) { return s + r.count; }, 0);

    var pm = T.match(/(\d{2}\.\d{2}\.\d{2})[^\d]*[~\-–][^\d]*(\d{2}\.\d{2}\.\d{2})/);

    return {
      period: pm ? (pm[1] + ' ~ ' + pm[2]) : '',
      kpi: kpi,
      failed: { total: total, reasons: reasons },
      savedBy: '씨마켓 자동동기화'
    };
  }

  function toast(msg, bad) {
    try {
      var d = document.createElement('div');
      d.textContent = msg;
      d.style.cssText = 'position:fixed;right:16px;bottom:16px;z-index:2147483647;padding:10px 15px;'
        + 'border-radius:10px;font-size:13px;font-weight:700;color:#fff;font-family:sans-serif;'
        + 'box-shadow:0 8px 24px rgba(0,0,0,.25);background:' + (bad ? '#dc2626' : '#16a34a');
      document.body.appendChild(d);
      setTimeout(function () { d.remove(); }, 4000);
    } catch (e) {}
  }

  function send(body) {
    var sig = JSON.stringify(body.kpi) + '|' + body.failed.total;
    var prev = null;
    try { prev = localStorage.getItem(SIG_LS); } catch (e) {}
    if (prev === sig) { log('값이 전과 같아 건너뜀', body.kpi); return; }

    var key = null;
    try { key = localStorage.getItem(KEY_LS); } catch (e) {}
    if (!key) {
      key = prompt('업무포털 수정 키(BID_EDIT_KEY)를 한 번만 입력해 주세요');
      if (!key) { log('키 입력 취소됨'); return; }
      try { localStorage.setItem(KEY_LS, key); } catch (e) {}
    }

    busy = true;
    log('보내는 중', body.kpi);

    fetch(PORTAL + '/api/bid', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-Edit-Key': key },
      body: JSON.stringify(body)
    })
      .then(function (r) { return r.json().then(function (j) { return { s: r.status, j: j }; }); })
      .then(function (res) {
        busy = false;
        if (res.j && res.j.success) {
          try { localStorage.setItem(SIG_LS, sig); } catch (e) {}
          log('저장 완료', res.j.kpi);
          toast('포털 갱신됨 · 진행중 ' + body.kpi.inProgress + ' / 마감 ' + body.kpi.closed);
        } else {
          log('실패', res.s, res.j);
          toast('포털 전송 실패: ' + ((res.j && res.j.message) || res.s), true);
          if (res.s === 401) { try { localStorage.removeItem(KEY_LS); } catch (e) {} }
        }
      })
      .catch(function (e) {
        busy = false;
        log('전송 오류', e.message);
        toast('포털 전송 실패: ' + e.message, true);
      });
  }

  function tick() {
    if (busy) return;
    var body = read();
    if (body) send(body);
  }

  log('시작 · 15초마다 「나의 거래 요약」 화면을 확인합니다');
  setTimeout(tick, 3000);          // 첫 확인
  setInterval(tick, 15000);        // 이후 계속

  // SPA 내부 이동(주소만 바뀌는 경우)도 잡습니다
  var lastUrl = location.href;
  setInterval(function () {
    if (location.href !== lastUrl) {
      lastUrl = location.href;
      log('화면 이동 감지', lastUrl);
      setTimeout(tick, 2500);
    }
  }, 1500);
})();
```

## 확인 방법

씨마켓 화면에서 `F12` → Console 에 `[씨마켓동기화]` 로 시작하는 줄이 보입니다.

| 콘솔 메시지 | 뜻 |
|---|---|
| `시작 · 15초마다…` | 스크립트가 돌고 있음 |
| `보내는 중 {inProgress:28,…}` | 숫자를 읽고 전송 시도 |
| `저장 완료` | **성공** (오른쪽 아래 초록 알림) |
| `값이 전과 같아 건너뜀` | 이미 보낸 값이라 생략 |
| `실패 401 …` | 수정 키가 Vercel 값과 다름 |
| `전송 오류 Failed to fetch` | 회사 네트워크가 vercel.app 차단 |
| 아무것도 안 나옴 | 스크립트 미설치·비활성 또는 `@match` 주소 불일치 |

**강제로 다시 보내기** (같은 값이라 건너뛸 때):
```javascript
localStorage.removeItem('fiti.bidSig'); location.reload();
```

## 더 튼튼하게 하려면

지금은 **화면 글자**를 읽습니다. 씨마켓이 문구를 바꾸면 깨질 수 있습니다.
`F12 → Network → Fetch/XHR` 에서 그 숫자가 담긴 **JSON 요청 주소**를 찾아 알려주시면,
글자 대신 그 API 를 직접 읽도록 바꿀 수 있습니다.
