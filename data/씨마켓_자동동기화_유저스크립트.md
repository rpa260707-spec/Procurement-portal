# 씨마켓 → 포털 자동 동기화 (유저스크립트)

북마클릿은 **누를 때만** 보냅니다. 이건 **씨마켓을 열기만 하면** 알아서 보냅니다.
평소처럼 씨마켓에 들어가기만 하면 포털 숫자가 따라옵니다. 누를 것도, 배포할 것도 없습니다.

> 씨마켓은 로그인 뒤에 있는 데이터라 **서버가 직접 못 읽습니다.**
> 그래서 이미 로그인해 있는 **내 브라우저**가 대신 읽어 보내는 구조입니다.
> 아이디·비밀번호는 어디에도 저장되지 않습니다.

## 설치 (한 번만, 3분)

1. 크롬 웹스토어에서 **Tampermonkey** 설치
2. Tampermonkey 아이콘 → **새 스크립트 추가**
3. 편집창 내용을 전부 지우고 아래 코드를 붙여넣기
4. **Ctrl + S** 로 저장

끝입니다. 이제 씨마켓 **나의 거래 요약** 화면에 들어갈 때마다 자동으로 포털에 보냅니다.

- 처음 한 번만 **수정 키**(`BID_EDIT_KEY`)를 물어봅니다
- 같은 값이면 보내지 않습니다 (쓸데없는 호출 방지)
- 보낸 뒤 오른쪽 아래에 작게 `포털 갱신됨` 이라고 1.5초 뜹니다

## 코드

```javascript
// ==UserScript==
// @name         씨마켓 → FITI 업무포털 동기화
// @namespace    fiti.procurement
// @version      1.0
// @description  씨마켓 나의 거래 요약 숫자를 업무포털로 자동 전송
// @match        https://www.c-market.net/*
// @run-at       document-idle
// @grant        none
// ==/UserScript==
(function () {
  'use strict';

  var PORTAL = 'https://fiti-procurement-portal.vercel.app';
  var KEY_LS = 'fiti.bidKey';
  var SIG_LS = 'fiti.bidSig';

  function num(s) {
    var m = String(s).replace(/,/g, '').match(/-?\d+/);
    return m ? parseInt(m[0], 10) : 0;
  }

  function read() {
    var T = document.body.innerText;
    if (T.indexOf('나의 거래 요약') < 0) return null;   // 그 화면이 아니면 아무것도 안 함

    function after(label) {
      var i = T.indexOf(label);
      if (i < 0) return 0;
      var m = T.slice(i + label.length, i + label.length + 40).match(/-?[\d,]+/);
      return m ? num(m[0]) : 0;
    }

    var kpi = {
      inProgress: after('진행중'),
      closed: after('전체 마감'),
      contracts: after('총 계약'),
      cardPending: after('카드 결제 대기'),
      invoices: after('계산서 및 영수증')
    };
    if (!kpi.closed && !kpi.contracts) return null;      // 아직 안 그려졌으면 대기

    var reasons = [], total = 0;
    var fi = T.indexOf('유찰 사유');
    if (fi >= 0) {
      var blk = T.slice(fi, fi + 1200).split('\n').map(function (x) { return x.trim(); }).filter(Boolean);
      var tm = blk.join(' ').match(/총\s*유찰\s*([\d,]+)\s*건/);
      if (tm) total = num(tm[1]);
      for (var i = 0; i < blk.length; i++) {
        var m = blk[i].match(/^(.+?)\s+([\d,]+)건\s*[·.]\s*[\d.]+%$/);
        if (m) { reasons.push({ label: m[1].trim(), count: num(m[2]) }); continue; }
        if (/건\s*[·.]\s*[\d.]+%/.test(blk[i]) && blk[i - 1] && !/유찰|총/.test(blk[i - 1])) {
          var c = blk[i].match(/([\d,]+)건/);
          if (c) reasons.push({ label: blk[i - 1], count: num(c[1]) });
        }
      }
    }
    if (!total) total = reasons.reduce(function (s, r) { return s + r.count; }, 0);

    var pm = T.match(/(\d{2}\.\d{2}\.\d{2})[^\d]*[~\-–][^\d]*(\d{2}\.\d{2}\.\d{2})/);

    return {
      period: pm ? (pm[1] + ' ~ ' + pm[2]) : '',
      kpi: kpi,
      failed: { total: total, reasons: reasons },
      savedBy: '씨마켓 자동동기화'
    };
  }

  function toast(msg, bad) {
    var d = document.createElement('div');
    d.textContent = msg;
    d.style.cssText = 'position:fixed;right:16px;bottom:16px;z-index:99999;padding:9px 14px;'
      + 'border-radius:10px;font-size:13px;font-weight:700;color:#fff;font-family:sans-serif;'
      + 'box-shadow:0 8px 24px rgba(0,0,0,.2);background:' + (bad ? '#dc2626' : '#16a34a');
    document.body.appendChild(d);
    setTimeout(function () { d.remove(); }, 2500);
  }

  function send(body) {
    var sig = JSON.stringify(body.kpi) + '|' + body.failed.total;
    try { if (localStorage.getItem(SIG_LS) === sig) return; } catch (e) {}   // 값이 같으면 건너뜀

    var key;
    try { key = localStorage.getItem(KEY_LS); } catch (e) {}
    if (!key) {
      key = prompt('업무포털 수정 키(BID_EDIT_KEY)를 한 번만 입력해 주세요');
      if (!key) return;
      try { localStorage.setItem(KEY_LS, key); } catch (e) {}
    }

    fetch(PORTAL + '/api/bid', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'X-Edit-Key': key },
      body: JSON.stringify(body)
    })
      .then(function (r) { return r.json(); })
      .then(function (j) {
        if (j && j.success) {
          try { localStorage.setItem(SIG_LS, sig); } catch (e) {}
          toast('포털 갱신됨 · 진행중 ' + body.kpi.inProgress + ' / 마감 ' + body.kpi.closed);
        } else {
          toast('포털 전송 실패: ' + (j && j.message ? j.message : ''), true);
          if (j && /키/.test(j.message || '')) { try { localStorage.removeItem(KEY_LS); } catch (e) {} }
        }
      })
      .catch(function (e) { toast('포털 전송 실패: ' + e.message, true); });
  }

  // 화면이 JS 로 그려지므로, 값이 나타날 때까지 잠깐 기다렸다가 한 번만 보냅니다.
  var tries = 0;
  var timer = setInterval(function () {
    tries++;
    var body = read();
    if (body) { clearInterval(timer); send(body); }
    else if (tries > 20) clearInterval(timer);   // 20초쯤 기다리고 포기
  }, 1000);
})();
```

## 잘 안 될 때

- **아무 일도 안 일어남** — 「나의 거래 요약」 화면인지 확인하세요. 다른 페이지에서는 일부러 동작하지 않습니다
- **숫자가 0** — 씨마켓 화면 문구가 바뀐 것입니다. 알려주시면 읽는 규칙을 고치겠습니다
- **「수정 키가 맞지 않습니다」** — 키가 틀렸습니다. 새로고침하면 다시 물어봅니다
- **「수정 키가 설정되지 않아…」** — Vercel 에 `BID_EDIT_KEY` 를 넣고 **Redeploy** 해야 합니다
- 한 번 보낸 값과 같으면 조용히 건너뜁니다. 강제로 다시 보내려면 씨마켓 화면에서 콘솔에
  `localStorage.removeItem('fiti.bidSig')` 후 새로고침

## 더 튼튼하게 하려면

지금은 **화면 글자**를 읽습니다. 씨마켓이 문구를 바꾸면 깨질 수 있습니다.
`F12 → Network → Fetch/XHR` 에서 그 숫자가 담긴 **JSON 요청 주소**를 찾아 알려주시면,
글자 대신 그 API 를 직접 읽도록 바꿀 수 있습니다. 그러면 화면이 바뀌어도 안 깨집니다.
