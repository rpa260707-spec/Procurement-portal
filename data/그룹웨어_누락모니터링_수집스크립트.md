# 입찰 누락 모니터링 — 그룹웨어 수집 스크립트

> 포털 「입찰구매 현황」 카드 아래 **누락 모니터링** 위젯에 들어갈 데이터를 만듭니다.

그룹웨어(`gw.fiti.re.kr`)는 로그인 쿠키가 있어야 읽히므로 **포털 서버가 직접 못 부릅니다.**
이미 로그인해 있는 **내 브라우저**가 화면을 읽어 포털로 보내는 구조입니다 — 씨마켓과 같은 방식입니다.

아이디·비밀번호는 어디에도 저장되지 않습니다.

---

## 무엇을 잡아내나

| 탭 | 뜻 | 판정 |
|---|---|---|
| **입찰 누락** | 품의만 올리고 입찰을 안 보냄 | 입찰 전송일시가 **비어 있음** |
| **마감 후 정체** | 마감일이 지났는데 아직 진행중 | 입찰 마감일시 **< 오늘** + 상태가 진행중 |
| **유찰 미처리** | 유찰인데 재공고도 취소도 안 함 | 진행단계 **유찰** + 다음 단계로 안 넘어감 |

각 건의 **경과일**을 같이 계산합니다. 포털에서 **10일 이상 적색 · 7일 이상 주황**으로 표시됩니다.

> **경과 14일이 넘은 건은 담지 않습니다 — 단 「유찰」은 예외로 전부 담습니다.**
> 그만큼 묵은 건은 따로 사유가 있어 손대지 않는 것이라, 목록에 남겨 두면 진짜 챙겨야 할 건이 묻힙니다.
> 수집 스크립트와 포털 API **양쪽에서** 거릅니다.

---

## 쓰는 법

1. 그룹웨어에서 **입찰구매관리(CDTS5122)** 를 엽니다
2. **입찰-소모품**으로 조회하고, 목록을 **한 화면에 다 나오게** 합니다
   (페이지당 건수를 최대로. 화면에 보이는 것만 읽습니다)
3. `F12` → **Console** 탭
4. 아래 코드를 붙여넣고 `Enter`
5. 읽어낸 숫자를 확인 창에서 보고 **확인**을 누르면 포털에 저장됩니다

처음 한 번은 **수정 키**를 물어봅니다. 포털 Vercel 환경변수 `BID_EDIT_KEY` 값이며,
씨마켓 동기화와 **같은 키**입니다. 한 번 넣으면 그 브라우저가 기억합니다.

> 크롬 콘솔이 붙여넣기를 막으면 `allow pasting` 을 입력한 뒤 다시 붙여넣으세요.

## 코드

```javascript
(function () {
  var PORTAL = 'https://fiti-procurement-portal.vercel.app';
  var KEY_LS = 'fiti.bidKey';
  var MAX_DAYS = 14;   // 경과 14일이 넘은 건은 뺍니다 — 따로 사유가 있는 건이라 보고 제외합니다

  // ── 표 찾기 : 머리글 이름으로 칸 위치를 잡습니다 (열 순서가 바뀌어도 따라갑니다)
  var WANT = {
    prNo:  ['구매번호'],
    bidNo: ['입찰번호'],
    item:  ['공고명', '품명', '건명'],
    step:  ['진행단계'],
    dept:  ['수요 부서', '수요부서', '부서'],
    emp:   ['수요 사원', '수요사원', '담당자', '신청자'],
    work:  ['구매업무'],
    sendDt:['입찰 전송일시', '입찰전송일시', '전송일시'],
    endDt: ['입찰 마감일시', '입찰마감일시', '마감일시'],
    state: ['입찰 상태', '입찰상태', '상태']
  };

  function txt(el) { return (el ? el.textContent : '').replace(/\s+/g, ' ').trim(); }

  // 머리글이 가장 많이 맞는 표를 고릅니다 (화면에 표가 여러 개일 수 있습니다)
  function pickTable() {
    var best = null, bestHit = 0;
    Array.prototype.forEach.call(document.querySelectorAll('table'), function (tb) {
      var hs = Array.prototype.map.call(tb.querySelectorAll('th'), txt);
      if (!hs.length) {
        var r0 = tb.querySelector('tr');
        hs = r0 ? Array.prototype.map.call(r0.children, txt) : [];
      }
      var hit = 0;
      Object.keys(WANT).forEach(function (k) {
        if (hs.some(function (h) { return WANT[k].some(function (w) { return h.indexOf(w) >= 0; }); })) hit++;
      });
      if (hit > bestHit) { bestHit = hit; best = { table: tb, heads: hs }; }
    });
    return bestHit >= 4 ? best : null;
  }

  function colMap(heads) {
    var m = {};
    Object.keys(WANT).forEach(function (k) {
      m[k] = -1;
      heads.forEach(function (h, i) {
        if (m[k] < 0 && WANT[k].some(function (w) { return h.indexOf(w) >= 0; })) m[k] = i;
      });
    });
    return m;
  }

  // '2026-09-28 17:00' · '2026.09.28' · '20260928' 을 모두 받습니다
  function toDate(v) {
    var s = String(v || '').trim();
    if (!s) return null;
    var m = s.match(/(\d{4})[.\-\/]?(\d{2})[.\-\/]?(\d{2})/);
    if (!m) return null;
    var d = new Date(+m[1], +m[2] - 1, +m[3]);
    return isNaN(d) ? null : d;
  }

  function daysAgo(d) {
    if (!d) return 0;
    var t = new Date(); t.setHours(0, 0, 0, 0);
    return Math.max(0, Math.round((t - d) / 86400000));
  }

  var found = pickTable();
  if (!found) {
    alert('입찰 목록 표를 찾지 못했습니다.\n\n입찰구매관리(CDTS5122) 목록 화면에서 실행해 주세요.\n그래도 안 되면 이 화면을 캡처해 전략구매팀에 알려 주세요.');
    return;
  }

  var col = colMap(found.heads);
  var rows = Array.prototype.slice.call(found.table.querySelectorAll('tbody tr'));
  if (!rows.length) rows = Array.prototype.slice.call(found.table.querySelectorAll('tr')).slice(1);

  var out = { notSent: [], stalled: [], failed: [] }, total = 0;
  var today = new Date(); today.setHours(0, 0, 0, 0);

  rows.forEach(function (tr) {
    var c = tr.children;
    function at(k) { return col[k] >= 0 && c[col[k]] ? txt(c[col[k]]) : ''; }

    var prNo = at('prNo');
    if (!prNo) return;                       // 소계·빈 줄 건너뜀
    total++;

    var rec = {
      prNo: prNo,
      item: at('item'),
      dept: at('dept'),
      emp:  at('emp'),
      endDt: at('endDt'),
      step: at('step')
    };

    var send = toDate(at('sendDt'));
    var end  = toDate(at('endDt'));
    var step = at('step');
    var state = at('state');
    var live = /진행/.test(step) || /진행/.test(state);

    // ① 입찰 누락 — 전송일시가 없음
    if (!send) {
      rec.days = daysAgo(end) || 0;
      if (rec.days <= MAX_DAYS) out.notSent.push(rec);
      return;
    }
    // ② 유찰 미처리 — 유찰인데 다음 단계로 안 넘어감
    if (/유찰/.test(step) && !/구매계약|발주|재공고|취소/.test(state)) {
      rec.days = daysAgo(end);
      out.failed.push(rec);                        // 유찰은 기간을 안 자릅니다
      return;
    }
    // ③ 마감 후 정체 — 마감일이 지났는데 진행중
    if (end && end < today && live) {
      rec.days = daysAgo(end);
      if (rec.days <= MAX_DAYS) out.stalled.push(rec);
    }
  });

  var msg = '포털에 보낼 값\n\n'
    + '읽은 줄 : ' + total + '건\n\n'
    + '입찰 누락      ' + out.notSent.length + '건\n'
    + '마감 후 정체   ' + out.stalled.length + '건\n'
    + '유찰 미처리    ' + out.failed.length + '건\n\n'
    + '보낼까요?';
  if (!confirm(msg)) { console.log('[누락모니터링] 취소됨', out); return; }

  var key = null;
  try { key = localStorage.getItem(KEY_LS); } catch (e) {}
  if (!key) {
    key = prompt('포털 수정 키(BID_EDIT_KEY)');
    if (!key) return;
    try { localStorage.setItem(KEY_LS, key); } catch (e) {}
  }

  fetch(PORTAL + '/api/bid-monitor', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'X-Edit-Key': key },
    body: JSON.stringify({
      scope: '입찰-소모품',
      totalRows: total,
      savedBy: '그룹웨어 수집',
      notSent: out.notSent,
      stalled: out.stalled,
      failed: out.failed
    })
  })
    .then(function (r) { return r.json(); })
    .then(function (j) {
      alert(j.success
        ? ('포털에 저장했습니다.\n\n입찰 누락 ' + j.counts.notSent
           + ' / 마감 후 정체 ' + j.counts.stalled
           + ' / 유찰 미처리 ' + j.counts.failed)
        : ('실패: ' + (j.message || '')));
      if (!j.success && /키/.test(j.message || '')) { try { localStorage.removeItem(KEY_LS); } catch (e) {} }
    })
    .catch(function (e) { alert('전송 실패: ' + e.message); });
})();
```

## 잘 안 될 때

| 보이는 것 | 원인 | 조치 |
|---|---|---|
| 「입찰 목록 표를 찾지 못했습니다」 | 목록 화면이 아니거나, 표가 일반 `<table>` 이 아님 | CDTS5122 목록에서 실행. 그래도 안 되면 **화면 캡처를 전략구매팀에** |
| 건수가 0 | 머리글 이름이 바뀜 | 캡처를 주시면 `WANT` 의 이름 목록에 추가합니다 |
| 「수정 키가 맞지 않습니다」 | 키 오타 | 다시 실행하면 새로 물어봅니다 |
| 「수정 키가 설정되지 않아…」 | Vercel 에 `BID_EDIT_KEY` 없음 | 환경변수 넣고 **Redeploy** |
| 전송 실패 (Failed to fetch) | 회사 네트워크가 vercel.app 차단 | 전산 문의 |

**읽은 줄이 화면에 보이는 건수뿐입니다.** 페이지가 나뉘어 있으면 그 페이지 것만 올라갑니다.
조회 조건에서 **페이지당 건수를 최대로** 놓고 실행하세요.

---

## 저장되는 것

개인정보는 담지 않습니다. **담당자는 이름만** 받고 사번·연락처·금액은 저장하지 않습니다.
탭당 **300건**까지, 경과일이 오래된 순으로 자릅니다.

저장소는 씨마켓과 같은 Vercel Blob 이고 파일명만 다릅니다 (`portal-bid-monitor.json`).
**배포 없이 갱신**되므로, 이 스크립트만 돌리면 포털이 바로 최신이 됩니다.
