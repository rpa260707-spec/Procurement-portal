# 그룹웨어 응답 엿보기 — 1단계 (누락 모니터링 자동화 준비)

> **이건 한 번만 씁니다.** 그룹웨어가 입찰-소모품 목록을 어떤 모양으로 받아 오는지 알아내려는 것입니다.
> 그 모양을 알면 **2단계 = 씨마켓처럼 저절로 도는 스크립트**를 만들 수 있습니다.
> 지금처럼 매번 복사·클릭하는 일은 그때 없어집니다.

**화면 글자를 긁지 않습니다.** 그룹웨어 목록은 `<table>` 이 아니라 그리드라 글자를 긁으면
칸 이름이 조금만 바뀌어도 깨집니다. 대신 **그리드가 서버에서 받아 오는 응답 자체**를 봅니다.
이쪽이 훨씬 튼튼합니다.

**아무것도 보내지 않습니다.** 이 스크립트는 읽기만 하고, 누를 때 **클립보드에 복사**만 합니다.
네트워크로 나가는 곳이 한 군데도 없습니다.

---

## 1. 크롬 「사용자 스크립트 허용」 — 이것부터

씨마켓 때와 같습니다. **이게 꺼져 있으면 아무 일도 안 일어납니다. 에러도 안 납니다.**

1. 주소창에 `chrome://extensions/`
2. **Tampermonkey** 카드 → **「세부정보」**
3. 아래로 내려 **「사용자 스크립트 허용」** 스위치 **켬**

## 2. 설치

1. Tampermonkey 아이콘 → **대시보드**
2. 왼쪽 위 **＋** (새 스크립트)
3. 편집창 전체 선택(`Ctrl+A`) → 삭제 → 아래 코드 붙여넣기 → `Ctrl+S`

## 3. 쓰는 법

```
1. 그룹웨어 → 구매관리 → 입찰구매관리(CDTS5122)
2. F5 로 한 번 새로 고침        ← 스크립트가 붙을 기회를 줍니다
3. 입찰-소모품 조회 (평소처럼)
4. 오른쪽 아래에 작은 상자가 뜹니다
       그룹웨어 응답 엿보기
       받은 응답 12개 · 목록으로 보이는 것 1개
       [ 목록 응답 복사 ]  [ 전부 복사 ]  [ 닫기 ]
5. 「목록 응답 복사」 누르고, 저한테 붙여넣어 주세요
```

**「목록으로 보이는 것 0개」 라고 나오면** 「전부 복사」를 눌러 주세요. 그걸 보고 찾겠습니다.

> 복사되는 내용에 **담당자 이름·수요부서**가 들어 있습니다. 사내 자료이니
> 이 대화에만 붙여넣으시고, 깃허브나 외부 메신저에는 올리지 마세요.
> 다 끝나면 이 스크립트는 **지우셔도 됩니다** (2단계 스크립트가 대신합니다).

## 코드

```javascript
// ==UserScript==
// @name         그룹웨어 응답 엿보기 (1단계)
// @namespace    fiti.procurement
// @version      1.0
// @description  입찰구매관리 그리드가 받아 오는 응답을 모아 클립보드로 복사합니다. 보내는 곳 없음.
// @match        *://gw.fiti.re.kr/*
// @run-at       document-start
// @grant        none
// ==/UserScript==

(function () {
  'use strict';

  /* 그리드는 화면을 그리기 전에 서버에서 자료를 받아 옵니다.
     그 통신을 가로채려면 페이지 코드보다 **먼저** 끼어들어야 해서 run-at document-start 입니다. */

  var CAP = [];            // 모아 둔 응답
  var MAX = 60;            // 너무 쌓이면 메모리를 먹으니 최근 것만
  /* 구매번호(CS + 숫자 10여 자리) 가 **몇 개나** 들어 있는지로 목록 응답을 가려냅니다.
     하나만 보고 고르면 화면 어딘가에 섞인 한 건짜리 응답이 뽑혀 버립니다. */
  var PRNO = /\bC[A-Z]\d{8,}\b/g;
  var SKIP = /(javascript|ecmascript|css|image|font)/i;   // 스크립트·스타일은 자료가 아닙니다

  function keep(url, method, req, text, ctype) {
    if (!text || text.length < 200) return;        // 빈 응답·핑은 버립니다
    if (SKIP.test(ctype || '')) return;
    PRNO.lastIndex = 0;
    var hits = (text.match(PRNO) || []).length;
    CAP.push({
      time: new Date().toTimeString().slice(0, 8),
      url: String(url || '').slice(0, 500),
      method: method || 'GET',
      ctype: ctype || '',
      req: String(req || '').slice(0, 3000),
      len: text.length,
      body: text.length > 300000 ? text.slice(0, 300000) + '\n…(잘림)' : text,
      hits: hits
    });
    if (CAP.length > MAX) CAP.shift();
    paint();
  }

  /* ── XMLHttpRequest ── */
  var OX = XMLHttpRequest.prototype.open;
  var OS = XMLHttpRequest.prototype.send;
  XMLHttpRequest.prototype.open = function (m, u) {
    this.__m = m; this.__u = u;
    return OX.apply(this, arguments);
  };
  XMLHttpRequest.prototype.send = function (b) {
    var self = this;
    try {
      this.addEventListener('load', function () {
        try {
          // responseType 이 blob/arraybuffer 면 responseText 가 터집니다
          var t = (self.responseType === '' || self.responseType === 'text') ? self.responseText : '';
          keep(self.__u, self.__m, (typeof b === 'string' ? b : ''), t,
               (self.getResponseHeader && self.getResponseHeader('content-type')) || '');
        } catch (e) { }
      });
    } catch (e) { }
    return OS.apply(this, arguments);
  };

  /* ── fetch ── */
  var OF = window.fetch;
  if (typeof OF === 'function') {
    window.fetch = function (input, init) {
      return OF.apply(this, arguments).then(function (r) {
        try {
          r.clone().text().then(function (t) {
            keep((r.url || input), (init && init.method) || 'GET',
                 (init && typeof init.body === 'string') ? init.body : '', t,
                 (r.headers && r.headers.get('content-type')) || '');
          })['catch'](function () { });
        } catch (e) { }
        return r;
      });
    };
  }

  /* ── 작은 상자 ── */
  var box, line;

  function build() {
    if (box || !document.body) return;
    box = document.createElement('div');
    box.setAttribute('style',
      'position:fixed;right:14px;bottom:14px;z-index:2147483647;' +
      'background:#fff;border:1px solid #c9d2e0;border-radius:10px;' +
      'box-shadow:0 6px 24px rgba(16,24,40,.18);padding:10px 12px;' +
      'font:12px/1.5 -apple-system,"Malgun Gothic",sans-serif;color:#101729;min-width:230px');
    box.innerHTML =
      '<div style="font-weight:800;margin-bottom:2px">그룹웨어 응답 엿보기</div>' +
      '<div id="fx-line" style="color:#5a6377;margin-bottom:7px">기다리는 중…</div>' +
      '<div style="display:flex;gap:5px;flex-wrap:wrap">' +
      '<button id="fx-one" type="button">목록 응답 복사</button>' +
      '<button id="fx-all" type="button">전부 복사</button>' +
      '<button id="fx-x"   type="button">닫기</button></div>';
    document.body.appendChild(box);
    var bs = box.querySelectorAll('button');
    for (var i = 0; i < bs.length; i++) {
      bs[i].setAttribute('style',
        'padding:4px 9px;border:1px solid #c9d2e0;border-radius:6px;' +
        'background:#f6f8fc;font:600 11.5px/1 inherit;cursor:pointer');
    }
    line = box.querySelector('#fx-line');
    box.querySelector('#fx-one').onclick = function () { copy(best()); };
    box.querySelector('#fx-all').onclick = function () { copy(CAP); };
    box.querySelector('#fx-x').onclick = function () { box.remove(); box = null; };
  }

  function paint() {
    build();
    if (!line) return;
    var n = 0;
    for (var i = 0; i < CAP.length; i++) if (CAP[i].hits > 0) n++;
    line.textContent = '받은 응답 ' + CAP.length + '개 · 목록으로 보이는 것 ' + n + '개';
  }

  /* 구매번호가 **많이** 들어 있는 것이 목록입니다. 수가 같으면 큰 쪽. */
  function best() {
    var list = CAP.filter(function (c) { return c.hits > 0; });
    if (!list.length) return [];
    list.sort(function (a, b) { return (b.hits - a.hits) || (b.len - a.len); });
    return list.slice(0, 2);
  }

  function copy(list) {
    if (!list || !list.length) {
      alert('목록으로 보이는 응답이 아직 없습니다.\n\n입찰-소모품을 한 번 조회한 뒤 다시 눌러 주세요.\n그래도 없으면 「전부 복사」를 눌러 주세요.');
      return;
    }
    var out = list.map(function (c, i) {
      return '===== 응답 ' + (i + 1) + ' =====\n'
        + 'time   : ' + c.time + '\n'
        + 'method : ' + c.method + '\n'
        + 'url    : ' + c.url + '\n'
        + 'type   : ' + c.ctype + '\n'
        + 'length : ' + c.len + '\n'
        + '--- 요청 본문 ---\n' + c.req + '\n'
        + '--- 응답 본문 ---\n' + c.body + '\n';
    }).join('\n');

    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(out).then(function () {
        alert('복사했습니다 (' + out.length.toLocaleString() + '자).\n\n대화창에 붙여넣어 주세요.');
      })['catch'](function () { showBox(out); });
    } else {
      showBox(out);
    }
  }

  /* 클립보드가 막히면 직접 고를 수 있게 펼쳐 줍니다 */
  function showBox(text) {
    var ta = document.createElement('textarea');
    ta.value = text;
    ta.setAttribute('style',
      'position:fixed;left:3vw;top:3vh;width:94vw;height:80vh;z-index:2147483647;' +
      'font:11px/1.4 monospace;padding:10px;border:2px solid #0f6b55;border-radius:8px');
    document.body.appendChild(ta);
    ta.focus(); ta.select();
    alert('자동 복사가 막혔습니다.\nCtrl+C 로 복사한 뒤 Esc 를 누르세요.');
    ta.addEventListener('keydown', function (e) { if (e.key === 'Escape') ta.remove(); });
  }

  /* 그룹웨어는 메뉴를 iframe 으로 엽니다. 모든 프레임에서 상자를 띄우면 서너 개가 겹칩니다.
     자료를 실제로 받은 프레임에서만 띄우고(= paint 가 부릅니다),
     6초가 지나도 아무것도 안 잡히면 맨 바깥 창에만 「기다리는 중」을 보여 줍니다. */
  setTimeout(function () {
    if (!CAP.length && window.top === window.self) build();
  }, 6000);
})();
```

```
설치 1회  →  그룹웨어 입찰구매관리 탭을 열어 두기만 하면

             · 응답을 가로채 입찰-소모품만 추려
             · 포털 /api/bid-monitor 로 바로 전송

```
