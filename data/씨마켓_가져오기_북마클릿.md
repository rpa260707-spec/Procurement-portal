# 씨마켓 입찰현황 자동 가져오기 (북마클릿)

씨마켓은 로그인 뒤에 있는 데이터라 서버가 직접 읽을 수 없습니다.
**이미 로그인해 있는 내 브라우저**에서 숫자를 읽어 포털로 보내는 방식입니다.
아이디·비밀번호는 어디에도 저장하지 않습니다.

## 쓰는 법

1. 아래 코드를 **즐겨찾기(북마크)** 로 하나 만듭니다
   - 즐겨찾기 막대에서 우클릭 → 추가 → 이름 `씨마켓→포털`, URL 칸에 아래 `javascript:...` 전체를 붙여넣기
2. 씨마켓에 로그인해서 **나의 거래 요약** 화면(`c-market.net/buyer/dashboard`)을 엽니다
3. 만들어 둔 **즐겨찾기를 클릭**합니다
4. 읽어낸 숫자를 확인 창에서 보고 **확인**을 누르면 포털에 저장됩니다

처음 한 번은 **수정 키**를 물어봅니다. 포털 Vercel 프로젝트의 환경변수 `BID_EDIT_KEY` 에
정해 둔 값을 넣으면 그 브라우저에 기억됩니다.

## 준비 (한 번만)

Vercel `procurement-portal` → Settings → Environment Variables

```
BID_EDIT_KEY = (아무 문자열, 팀 안에서만 공유)
```

저장 후 **Redeploy**. 이 키가 없으면 포털은 **읽기 전용**으로만 동작합니다(아무나 숫자를 바꾸지 못하게).

---

## 북마클릿 코드

```javascript
javascript:(function(){
  var PORTAL='https://fiti-procurement-portal.vercel.app';
  function num(s){var m=String(s).replace(/,/g,'').match(/-?\d+/);return m?parseInt(m[0],10):0;}
  var T=document.body.innerText;
  function after(label){
    var i=T.indexOf(label); if(i<0) return 0;
    var m=T.slice(i+label.length, i+label.length+40).match(/-?[\d,]+/);
    return m?num(m[0]):0;
  }
  var kpi={
    inProgress:after('진행중'), closed:after('전체 마감'), contracts:after('총 계약'),
    cardPending:after('카드 결제 대기'), invoices:after('계산서 및 영수증')
  };
  var reasons=[], total=0;
  var fi=T.indexOf('유찰 사유');
  if(fi>=0){
    var blk=T.slice(fi, fi+1200).split('\n').map(function(x){return x.trim();}).filter(Boolean);
    var tm=blk.join(' ').match(/총\s*유찰\s*([\d,]+)\s*건/); if(tm) total=num(tm[1]);
    for(var i=0;i<blk.length;i++){
      var m=blk[i].match(/^(.+?)\s+([\d,]+)건\s*[·.]\s*[\d.]+%$/);
      if(m) reasons.push({label:m[1].trim(), count:num(m[2])});
      else if(/건\s*[·.]\s*[\d.]+%/.test(blk[i]) && blk[i-1] && !/유찰|총/.test(blk[i-1])){
        var c=blk[i].match(/([\d,]+)건/); if(c) reasons.push({label:blk[i-1], count:num(c[1])});
      }
    }
  }
  if(!total) total=reasons.reduce(function(s,r){return s+r.count;},0);
  var pm=T.match(/(\d{2}\.\d{2}\.\d{2})[^\d]*[~\-–][^\d]*(\d{2}\.\d{2}\.\d{2})/);
  var period=pm?(pm[1]+' ~ '+pm[2]):'';

  var body={period:period, kpi:kpi, failed:{total:total, reasons:reasons}, savedBy:'씨마켓 북마클릿'};
  var msg='포털에 보낼 값\n\n'
    +'기간 : '+(period||'(못 읽음)')+'\n'
    +'진행중 '+kpi.inProgress+' / 전체 마감 '+kpi.closed+' / 총 계약 '+kpi.contracts+'\n'
    +'카드 결제 대기 '+kpi.cardPending+' / 계산서·영수증 '+kpi.invoices+'\n\n'
    +'유찰 '+total+'건\n'+reasons.map(function(r){return ' · '+r.label+' '+r.count+'건';}).join('\n')
    +'\n\n보낼까요?';
  if(!confirm(msg)) return;

  var key=localStorage.getItem('fiti.bidKey');
  if(!key){ key=prompt('포털 수정 키(BID_EDIT_KEY)'); if(!key) return; localStorage.setItem('fiti.bidKey',key); }

  fetch(PORTAL+'/api/bid',{method:'POST',headers:{'Content-Type':'application/json','X-Edit-Key':key},body:JSON.stringify(body)})
    .then(function(r){return r.json();})
    .then(function(j){ alert(j.success?'포털에 저장했습니다.':('실패: '+(j.message||'')));
                       if(!j.success && /키/.test(j.message||'')) localStorage.removeItem('fiti.bidKey'); })
    .catch(function(e){ alert('전송 실패: '+e.message); });
})();
```

## 잘 안 될 때

- **숫자가 0으로 나옴** — 씨마켓 화면 글자가 바뀐 경우입니다. 확인 창에서 보고 **취소**한 뒤 알려 주세요. 읽는 규칙을 고치면 됩니다
- **「수정 키가 맞지 않습니다」** — 키를 잘못 넣은 것입니다. 다시 클릭하면 새로 물어봅니다
- **「수정 키가 설정되지 않아…」** — Vercel 에 `BID_EDIT_KEY` 를 넣고 Redeploy 해야 합니다
- 그래도 안 되면 `data/bid.json` 을 직접 고쳐 배포하는 기존 방식이 그대로 살아 있습니다
